- [ ] **Step 3: Implement `MainThreadDispatcher`**

Create `crates/dat0-app/src/main_bridge.rs`:

```rust
//! `MainThreadDispatcher` — post a `FnOnce(&mut AppProxy) + Send` closure
//! onto the GPUI main thread from any thread.
//!
//! P3a PD-010 closure: `AsyncApp::update` calls `RefCell::borrow_mut` on
//! state owned by the Cocoa main thread; calling it from a tokio worker
//! panics. This module replaces that direct path with a
//! `futures::channel::mpsc::unbounded` channel whose receiver lives in a
//! `cx.spawn` future registered during app init. The sender is `Send` +
//! cloneable; tokio tasks (e.g., the UDS forwarder, the import-cancel
//! handler) post closures into it.
//!
//! See `docs/internal/gpui-api-notes.md` § "Globals + cross-thread
//! dispatch" for the verified `cx.update` / `cx.spawn` shape this module
//! relies on (T0 spike).

use std::any::Any;

use futures::channel::mpsc;
use futures::SinkExt;
use futures::StreamExt;

/// Closure shape: takes the current `App` proxy and mutates state on the
/// main thread. The proxy type is a re-export of GPUI's `App` (see
/// `gpui-api-notes.md`); aliased here to keep dependents from importing
/// the GPUI surface directly.
pub type AppProxy = gpui::App;
pub type MainClosure = Box<dyn FnOnce(&mut AppProxy) + Send + 'static>;

/// Sender side. Cheap to clone; safe to send across threads.
#[derive(Clone)]
pub struct MainThreadDispatcher {
    tx: mpsc::UnboundedSender<MainClosure>,
}

/// Receiver side. Lives in a `cx.spawn` future during app init; runs
/// until the last `MainThreadDispatcher` is dropped.
pub struct MainLoop {
    rx: mpsc::UnboundedReceiver<MainClosure>,
}

#[derive(Debug, thiserror::Error)]
pub enum DispatchError {
    #[error("receiver loop is closed")]
    Closed,
}

impl MainThreadDispatcher {
    /// Create a new dispatcher + matching main-loop receiver.
    pub fn new() -> (Self, MainLoop) {
        let (tx, rx) = mpsc::unbounded();
        (Self { tx }, MainLoop { rx })
    }

    /// Post a closure to run on the main thread.
    pub fn dispatch<F>(&self, f: F) -> Result<(), DispatchError>
    where
        F: FnOnce(&mut AppProxy) + Send + 'static,
    {
        let mut tx = self.tx.clone();
        // `unbounded_send` is sync + non-blocking; we use it via the
        // `Sink::send` API for parity with bounded channels.
        tx.unbounded_send(Box::new(f) as MainClosure)
            .map_err(|_| DispatchError::Closed)
    }
}

impl MainLoop {
    /// Test-only drain. Pops each pending closure and invokes it with
    /// a synthetic shim — there is no real `&mut App` available in
    /// unit tests, so the shim is created via `unsafe { std::mem::zeroed() }`
    /// inside a closure that **must not dereference the argument**.
    /// Test closures verify side-channel state (atomics, Arc<Mutex<_>>)
    /// they captured before dispatch; they never read the `&mut App`.
    ///
    /// Production uses `consume(&mut AsyncApp)`; that path always has a
    /// real `App` and is exercised by `tests/single_instance.rs`.
    #[cfg(any(test, doctest))]
    pub fn drain_for_test(mut self) {
        futures::executor::block_on(async move {
            while let Some(f) = self.rx.next().await {
                // Safety: test closures never dereference the &mut App;
                // they capture side-channel state instead. The pointer
                // is never followed.
                let shim: *mut AppProxy = std::ptr::NonNull::<AppProxy>::dangling().as_ptr();
                // SAFETY: see contract above. Reviewer: any test that
                // dereferences the `&mut App` argument is UB.
                unsafe { f(&mut *shim) };
            }
        });
    }

    /// Consume the receiver inside a GPUI `cx.spawn` future. Each
    /// received closure runs on the main thread via `cx.update`.
    ///
    /// Call shape (verified at T0 spike):
    ///
    /// ```ignore
    /// cx.spawn(|mut cx| async move {
    ///     main_loop.consume(&mut cx).await
    /// }).detach();
    /// ```
    pub async fn consume(mut self, cx: &mut gpui::AsyncApp) -> Result<(), anyhow::Error> {
        while let Some(f) = self.rx.next().await {
            cx.update(|app| f(app))?;
        }
        Ok(())
    }
}
```

The `drain_for_test` shim sidesteps the chicken-and-egg of needing an `&mut App` in a unit test that has no GPUI app running. Test closures use side-channel state (atomics, `Arc<Mutex<_>>`); they must not dereference the argument. T0 spike checks whether `gpui::App::test()` (or similar test seam) exists — if it does, replace the unsafe shim with the safe test constructor and remove this caveat.

        Ok(mut q) => q.push(banner),
        Err(poisoned) => {

use std::collections::HashMap;
use std::sync::Arc;

use parking_lot::RwLock;
use thiserror::Error;

/// Stable id for an action. Used by the palette + Banner action lookups.
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct ActionId(String);

impl ActionId {
    pub fn from(s: impl Into<String>) -> Self {
        Self(s.into())
    }
    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl std::fmt::Display for ActionId {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.write_str(&self.0)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ActionGroup {
    Navigation,
    Theme,
    File,
    Settings,
    Recovery,
    Import,
}

/// Closure type for action dispatch. Must be `Send + Sync` because the
/// registry is `Arc<RwLock<...>>`; actions that capture non-`Send` state
/// route through `MainThreadDispatcher` (T1) inside the closure.
pub type DispatchFn = Arc<dyn Fn(&mut gpui::App) + Send + Sync + 'static>;

#[derive(Clone)]
pub struct ActionDescriptor {
    pub id: ActionId,
    pub title: String,
    pub group: ActionGroup,
    pub keybinding: Option<gpui::Keystroke>,
    pub dispatch: DispatchFn,
}

#[derive(Debug, Error)]
pub enum RegisterError {
    #[error("duplicate action id: {0}")]
    DuplicateId(String),
}

#[derive(Clone, Default)]
pub struct ActionRegistry {
    inner: Arc<RwLock<HashMap<ActionId, ActionDescriptor>>>,
}

impl ActionRegistry {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn register(&self, desc: ActionDescriptor) -> Result<(), RegisterError> {
        let mut w = self.inner.write();
        if w.contains_key(&desc.id) {
            return Err(RegisterError::DuplicateId(desc.id.to_string()));
        }
        w.insert(desc.id.clone(), desc);
        Ok(())
    }

    pub fn get(&self, id: &ActionId) -> Option<ActionDescriptor> {
        self.inner.read().get(id).cloned()
    }

    pub fn iter(&self) -> impl Iterator<Item = ActionDescriptor> {
        let snapshot: Vec<ActionDescriptor> = self.inner.read().values().cloned().collect();
        snapshot.into_iter()
    }

    pub fn count(&self) -> usize {
        self.inner.read().len()
    }
}

pub fn orphan_scan_emit(scratch_root: &std::path::Path) -> Vec<crate::error_ux::Banner> {
    let mut count = 0usize;
    if let Ok(entries) = std::fs::read_dir(scratch_root) {
        for e in entries.flatten() {
            if e.path().join("session.json").is_file() {
                count += 1;
            }

    if count > 0 {

                format!("{count} previous session{} found", if count == 1 { "" } else { "s" }),
                "Restore tabs or discard them.".into(),

        );
    }
    for b in &banners {
        crate::error_ux::push(b.clone());
    }

use crate::actions::registry::{ActionDescriptor, ActionRegistry};

/// Filter actions by a fuzzy subsequence match against the title.
/// Case-insensitive; preserves ordering by registration.
pub fn filter(reg: &ActionRegistry, query: &str) -> Vec<ActionDescriptor> {
    let q = query.trim().to_lowercase();
    if q.is_empty() {
        return reg.iter().collect();
    }
    reg.iter()
        .filter(|d| subsequence_match(&d.title.to_lowercase(), &q))
        .collect()
}

fn subsequence_match(haystack: &str, needle: &str) -> bool {
    let mut hay = haystack.chars();
    for c in needle.chars() {
        match hay.find(|h| h.eq_ignore_ascii_case(&c)) {
            Some(_) => continue,
            None => return false,
        }
    }
    true
}

```rust
use std::io::Write;

use anyhow::{anyhow, Context};
use sha2::{Digest, Sha256};

/// Fetch a remote sample to `<state_root>/samples/<filename>`. Verifies
/// the SHA256 against `expected_sha_hex` after download. Returns the
/// cached path on success.
///
/// If the file already exists at the cache path, returns the existing
/// path without verifying (caller is responsible for invalidation on
/// version bumps — the URL includes a version segment).
pub async fn fetch_remote(
    url: &str,
    expected_sha_hex: &str,
    state_root: &Path,
    dest_filename: &str,
) -> anyhow::Result<PathBuf> {
    let dir = state_root.join("samples");
    std::fs::create_dir_all(&dir).context("create samples dir")?;
    let dest = dir.join(dest_filename);
    if dest.exists() {
        return Ok(dest);
    }

    let resp = reqwest::get(url).await.with_context(|| format!("GET {url}"))?;
    if !resp.status().is_success() {
        return Err(anyhow!("HTTP status {} for {url}", resp.status()));
    }
    let bytes = resp.bytes().await.context("read response body")?;

    let mut h = Sha256::new();
    h.update(&bytes);
    let got_sha = format!("{:x}", h.finalize());
    if got_sha != expected_sha_hex.to_lowercase() {
        return Err(anyhow!(
            "checksum mismatch: expected {} got {} for {url}",
            expected_sha_hex, got_sha,
        ));
    }

    // Atomic-write: write to temp file in same dir, then rename.
    let tmp_path = dir.join(format!("{dest_filename}.part"));
    {
        let mut f = std::fs::File::create(&tmp_path).context("create temp file")?;
        f.write_all(&bytes).context("write bytes")?;
        f.sync_all().context("fsync")?;
    }
    std::fs::rename(&tmp_path, &dest).context("rename into place")?;

    Ok(dest)
}

        "Sample data download failed",
        format!("Couldn't download from {url}: {err}"),

}

    pub cancel: Arc<AtomicBool>,
    pub bytes_done: Arc<AtomicU64>,
    pub total_bytes: u64,
    /// The DuckDB statement handle so `engine.interrupt(handle)` can be
    /// called on cancel. P2 exposes this via `engine.current_handle()`
    /// or similar — verify against the engine surface and store it
    /// here. Boxing as `Any` to keep this module independent of the
    /// engine internals.
    pub handle: Arc<Mutex<Option<u64>>>, // placeholder repr; replace with the real handle type
}

impl ImportProgress {
    pub fn new(total_bytes: u64) -> Self {
        Self {
            cancel: Arc::new(AtomicBool::new(false)),
            bytes_done: Arc::new(AtomicU64::new(0)),
            total_bytes,
            handle: Arc::new(Mutex::new(None)),
        }
    }

    pub fn update(&self, done: u64) {
        self.bytes_done.store(done, Ordering::SeqCst);
    }

    pub fn bytes_done(&self) -> u64 {
        self.bytes_done.load(Ordering::SeqCst)
    }

    pub fn total_bytes(&self) -> u64 {
        self.total_bytes
    }

    pub fn is_cancelled(&self) -> bool {
        self.cancel.load(Ordering::SeqCst)
    }
}

/// One active import per app (a P3b simplification — multi-window
/// extension is a P3c concern). Updated by `file_drop` when an import
/// starts and cleared when it finishes / cancels.
static ACTIVE: Lazy<Mutex<Option<ImportProgress>>> = Lazy::new(|| Mutex::new(None));

pub fn set_active(p: ImportProgress) {
    *ACTIVE.lock() = Some(p);
}

pub fn clear_active() {
    *ACTIVE.lock() = None;
}

pub fn active() -> Option<ImportProgress> {
    ACTIVE.lock().clone()
}

/// Sets the cancel flag on the given progress and (later) calls

pub fn request_cancel(cancel: &Arc<AtomicBool>) {
    cancel.store(true, Ordering::SeqCst);

}

pub fn cancel_active(_app: &mut gpui::App) {

        request_cancel(&p.cancel);
        if let Some(_h) = *p.handle.lock() {

        }

    }
}
